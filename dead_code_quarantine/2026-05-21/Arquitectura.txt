// ===== MODELO DEL CARRITO =====
data class CartItem(
    val product: Product,
    var quantity: Int,
    var discount: BigDecimal = BigDecimal.ZERO,
    var discountType: DiscountType = DiscountType.NONE // PERCENTAGE, FIXED, NONE
) {
    val subtotal: BigDecimal
        get() = product.price * quantity.toBigDecimal()

    val discountAmount: BigDecimal
        get() = when (discountType) {
            DiscountType.PERCENTAGE -> subtotal * discount / BigDecimal(100)
            DiscountType.FIXED -> discount
            DiscountType.NONE -> BigDecimal.ZERO
        }

    val taxAmount: BigDecimal
        get() = (subtotal - discountAmount) * product.taxRate / BigDecimal(100)

    val total: BigDecimal
        get() = subtotal - discountAmount + taxAmount
}

@HiltViewModel
class SaleViewModel @Inject constructor(
    private val createSaleUseCase: CreateSaleUseCase,
    private val getProductByBarcodeUseCase: GetProductByBarcodeUseCase,
    private val printReceiptUseCase: PrintReceiptUseCase,
    private val cashRegisterRepository: CashRegisterRepository
) : ViewModel() {

    private val _cartItems = MutableStateFlow<List<CartItem>>(emptyList())
    val cartItems: StateFlow<List<CartItem>> = _cartItems.asStateFlow()

    private val _cartSummary = MutableStateFlow(CartSummary())
    val cartSummary: StateFlow<CartSummary> = _cartSummary.asStateFlow()

    private val _saleState = MutableStateFlow<SaleState>(SaleState.Idle)
    val saleState: StateFlow<SaleState> = _saleState.asStateFlow()

    fun addProduct(product: Product, quantity: Int = 1) {
        val currentItems = _cartItems.value.toMutableList()
        val existingItem = currentItems.find { it.product.id == product.id }

        if (existingItem != null) {
            // Verificar stock
            if (existingItem.quantity + quantity > product.stock) {
                _saleState.value = SaleState.Error("Stock insuficiente")
                return
            }
            existingItem.quantity += quantity
        } else {
            if (quantity > product.stock) {
                _saleState.value = SaleState.Error("Stock insuficiente")
                return
            }
            currentItems.add(CartItem(product, quantity))
        }

        _cartItems.value = currentItems
        recalculateSummary()
    }

    fun removeProduct(productId: String) {
        _cartItems.value = _cartItems.value.filter { it.product.id != productId }
        recalculateSummary()
    }

    fun updateQuantity(productId: String, newQuantity: Int) {
        if (newQuantity <= 0) {
            removeProduct(productId)
            return
        }
        _cartItems.value = _cartItems.value.map {
            if (it.product.id == productId) it.copy(quantity = newQuantity) else it
        }
        recalculateSummary()
    }

    fun clearCart() {
        _cartItems.value = emptyList()
        recalculateSummary()
    }

    private fun recalculateSummary() {
        val items = _cartItems.value
        _cartSummary.value = CartSummary(
            itemCount = items.sumOf { it.quantity },
            subtotal = items.sumOf { it.subtotal },
            discountTotal = items.sumOf { it.discountAmount },
            taxTotal = items.sumOf { it.taxAmount },
            total = items.sumOf { it.total }
        )
    }

    fun processSale(
        paymentMethodId: String,
        amountPaid: BigDecimal,
        customerId: String? = null,
        notes: String? = null
    ) {
        viewModelScope.launch {
            _saleState.value = SaleState.Processing
            try {
                val result = createSaleUseCase(
                    items = _cartItems.value,
                    summary = _cartSummary.value,
                    paymentMethodId = paymentMethodId,
                    amountPaid = amountPaid,
                    customerId = customerId,
                    notes = notes
                )
                result.onSuccess { sale ->
                    _saleState.value = SaleState.Success(sale)
                    clearCart()
                }.onFailure { error ->
                    _saleState.value = SaleState.Error(error.message ?: "Error desconocido")
                }
            } catch (e: Exception) {
                _saleState.value = SaleState.Error(e.message ?: "Error al procesar venta")
            }
        }
    }

    fun searchByBarcode(barcode: String) {
        viewModelScope.launch {
            getProductByBarcodeUseCase(barcode)?.let { product ->
                addProduct(product)
            } ?: run {
                _saleState.value = SaleState.Error("Producto no encontrado")
            }
        }
    }
}

data class CartSummary(
    val itemCount: Int = 0,
    val subtotal: BigDecimal = BigDecimal.ZERO,
    val discountTotal: BigDecimal = BigDecimal.ZERO,
    val taxTotal: BigDecimal = BigDecimal.ZERO,
    val total: BigDecimal = BigDecimal.ZERO
)

sealed class SaleState {
    object Idle : SaleState()
    object Processing : SaleState()
    data class Success(val sale: Sale) : SaleState()
    data class Error(val message: String) : SaleState()
}

// ===== SYNC MANAGER =====
@Singleton
class SyncManager @Inject constructor(
    private val syncQueueDao: SyncQueueDao,
    private val apiService: PosApiService,
    private val networkMonitor: NetworkMonitor,
    private val workManager: WorkManager
) {
    fun enqueueSync(entityType: String, entityId: String, action: String, payload: String) {
        CoroutineScope(Dispatchers.IO).launch {
            syncQueueDao.insert(
                SyncQueue(
                    entityType = entityType,
                    entityId = entityId,
                    action = action,
                    payload = payload
                )
            )
            // Intentar sincronizar inmediatamente si hay conexión
            if (networkMonitor.isConnected()) {
                triggerImmediateSync()
            }
        }
    }

    fun schedulePeriodicalSync() {
        val constraints = Constraints.Builder()
            .setRequiredNetworkType(NetworkType.CONNECTED)
            .build()

        val syncWork = PeriodicWorkRequestBuilder<SyncWorker>(
            15, TimeUnit.MINUTES
        )
            .setConstraints(constraints)
            .setBackoffCriteria(BackoffPolicy.EXPONENTIAL, 1, TimeUnit.MINUTES)
            .build()

        workManager.enqueueUniquePeriodicWork(
            "pos_sync",
            ExistingPeriodicWorkPolicy.KEEP,
            syncWork
        )
    }

    private fun triggerImmediateSync() {
        val constraints = Constraints.Builder()
            .setRequiredNetworkType(NetworkType.CONNECTED)
            .build()

        val syncWork = OneTimeWorkRequestBuilder<SyncWorker>()
            .setConstraints(constraints)
            .build()

        workManager.enqueue(syncWork)
    }
}

// ===== SYNC WORKER =====
@HiltWorker
class SyncWorker @AssistedInject constructor(
    @Assisted context: Context,
    @Assisted params: WorkerParameters,
    private val syncQueueDao: SyncQueueDao,
    private val apiService: PosApiService
) : CoroutineWorker(context, params) {

    override suspend fun doWork(): Result {
        val pendingItems = syncQueueDao.getPendingItems()

        if (pendingItems.isEmpty()) return Result.success()

        var allSuccess = true

        for (item in pendingItems) {
            try {
                syncQueueDao.updateStatus(item.id, SyncStatus.IN_PROGRESS)

                val response = when (item.entityType) {
                    "sale" -> syncSale(item)
                    "product" -> syncProduct(item)
                    "customer" -> syncCustomer(item)
                    else -> false
                }

                if (response) {
                    syncQueueDao.updateStatus(item.id, SyncStatus.COMPLETED)
                } else {
                    handleFailure(item)
                    allSuccess = false
                }
            } catch (e: Exception) {
                handleFailure(item, e.message)
                allSuccess = false
            }
        }

        return if (allSuccess) Result.success() else Result.retry()
    }

    private suspend fun handleFailure(item: SyncQueue, error: String? = null) {
        val newAttempts = item.attempts + 1
        if (newAttempts >= item.maxAttempts) {
            syncQueueDao.updateStatus(item.id, SyncStatus.FAILED)
        } else {
            syncQueueDao.updateAttempt(item.id, newAttempts, error)
        }
    }
}

// ===== IMPLEMENTACIÓN BLUETOOTH (ESC/POS) =====
class BluetoothEscPosPrinter @Inject constructor(
    private val context: Context
) : ReceiptPrinter {

    private var socket: BluetoothSocket? = null
    private var outputStream: OutputStream? = null

    override suspend fun connect(): Boolean = withContext(Dispatchers.IO) {
        try {
            // Obtener dispositivo emparejado
            val device = getStoredBluetoothDevice() ?: return@withContext false
            val uuid = UUID.fromString("00001101-0000-1000-8000-00805F9B34FB")
            socket = device.createRfcommSocketToServiceRecord(uuid)
            socket?.connect()
            outputStream = socket?.outputStream
            true
        } catch (e: Exception) {
            Log.e("Printer", "Error connecting", e)
            false
        }
    }

    override suspend fun printReceipt(receipt: Receipt): PrintResult {
        return try {
            val os = outputStream ?: return PrintResult.Error("No conectado")

            // Comandos ESC/POS
            os.write(ESC_INIT)                    // Inicializar
            os.write(ESC_CENTER)                   // Centrar
            os.write(ESC_BOLD_ON)                  // Negrita
            os.write("${receipt.businessName}\n".toByteArray())
            os.write(ESC_BOLD_OFF)
            os.write("${receipt.businessAddress}\n".toByteArray())
            os.write("Tel: ${receipt.businessPhone}\n".toByteArray())
            os.write("RFC: ${receipt.businessTaxId}\n".toByteArray())
            os.write(SEPARATOR)

            os.write(ESC_LEFT)                     // Alinear izquierda
            os.write("Venta #: ${receipt.saleNumber}\n".toByteArray())
            os.write("Fecha: ${receipt.date}\n".toByteArray())
            os.write("Cajero: ${receipt.cashierName}\n".toByteArray())
            receipt.customerName?.let {
                os.write("Cliente: $it\n".toByteArray())
            }
            os.write(SEPARATOR)

            // Items
            os.write(formatHeader().toByteArray())
            receipt.items.forEach { item ->
                os.write(formatItem(item).toByteArray())
            }
            os.write(SEPARATOR)

            // Totales
            os.write(ESC_RIGHT)
            os.write("Subtotal: ${receipt.subtotal}\n".toByteArray())
            if (receipt.discountTotal != "$0.00") {
                os.write("Descuento: -${receipt.discountTotal}\n".toByteArray())
            }
            os.write("Impuesto: ${receipt.taxTotal}\n".toByteArray())
            os.write(ESC_BOLD_ON)
            os.write(ESC_DOUBLE_HEIGHT)
            os.write("TOTAL: ${receipt.total}\n".toByteArray())
            os.write(ESC_NORMAL_SIZE)
            os.write(ESC_BOLD_OFF)

            os.write("\nPago: ${receipt.paymentMethod}\n".toByteArray())
            os.write("Recibido: ${receipt.amountPaid}\n".toByteArray())
            os.write("Cambio: ${receipt.changeAmount}\n".toByteArray())

            // Pie de página
            os.write(ESC_CENTER)
            os.write("\n${receipt.footerMessage ?: "¡Gracias por su compra!"}\n\n\n".toByteArray())

            cutPaper()
            os.flush()

            PrintResult.Success
        } catch (e: Exception) {
            PrintResult.Error(e.message ?: "Error de impresión")
        }
    }

    companion object {
        val ESC_INIT = byteArrayOf(0x1B, 0x40)
        val ESC_CENTER = byteArrayOf(0x1B, 0x61, 0x01)
        val ESC_LEFT = byteArrayOf(0x1B, 0x61, 0x00)
        val ESC_RIGHT = byteArrayOf(0x1B, 0x61, 0x02)
        val ESC_BOLD_ON = byteArrayOf(0x1B, 0x45, 0x01)
        val ESC_BOLD_OFF = byteArrayOf(0x1B, 0x45, 0x00)
        val ESC_DOUBLE_HEIGHT = byteArrayOf(0x1D, 0x21, 0x11)
        val ESC_NORMAL_SIZE = byteArrayOf(0x1D, 0x21, 0x00)
        val ESC_CUT = byteArrayOf(0x1D, 0x56, 0x00)
        val SEPARATOR = "================================\n".toByteArray()
    }
}

class BarcodeScannerManager @Inject constructor() {

    private val scanner = BarcodeScanning.getClient(
        BarcodeScannerOptions.Builder()
            .setBarcodeFormats(
                Barcode.FORMAT_EAN_13,
                Barcode.FORMAT_EAN_8,
                Barcode.FORMAT_UPC_A,
                Barcode.FORMAT_UPC_E,
                Barcode.FORMAT_CODE_128,
                Barcode.FORMAT_CODE_39,
                Barcode.FORMAT_QR_CODE
            )
            .build()
    )

    fun getAnalyzer(onBarcodeDetected: (String) -> Unit): ImageAnalysis.Analyzer {
        return ImageAnalysis.Analyzer { imageProxy ->
            val mediaImage = imageProxy.image
            if (mediaImage != null) {
                val inputImage = InputImage.fromMediaImage(
                    mediaImage,
                    imageProxy.imageInfo.rotationDegrees
                )

                scanner.process(inputImage)
                    .addOnSuccessListener { barcodes ->
                        barcodes.firstOrNull()?.rawValue?.let { value ->
                            onBarcodeDetected(value)
                        }
                    }
                    .addOnCompleteListener {
                        imageProxy.close()
                    }
            }
        }
    }
}

@Dao
interface ReportDao {

    // Ventas del día
    @Query("""
        SELECT * FROM sales 
        WHERE createdAt BETWEEN :startOfDay AND :endOfDay 
        AND status = 'COMPLETED'
        ORDER BY createdAt DESC
    """)
    fun getSalesToday(startOfDay: Long, endOfDay: Long): Flow<List<Sale>>

    // Resumen de ventas por período
    @Query("""
        SELECT 
            COUNT(*) as totalTransactions,
            SUM(total) as totalRevenue,
            SUM(taxTotal) as totalTax,
            SUM(discountTotal) as totalDiscounts,
            AVG(total) as averageTicket
        FROM sales 
        WHERE createdAt BETWEEN :startDate AND :endDate
        AND status = 'COMPLETED'
    """)
    suspend fun getSalesSummary(startDate: Long, endDate: Long): SalesSummary

    // Top productos más vendidos
    @Query("""
        SELECT 
            sd.productId,
            sd.productName,
            SUM(sd.quantity) as totalQuantity,
            SUM(sd.total) as totalRevenue
        FROM sale_details sd
        INNER JOIN sales s ON s.id = sd.saleId
        WHERE s.createdAt BETWEEN :startDate AND :endDate
        AND s.status = 'COMPLETED'
        GROUP BY sd.productId
        ORDER BY totalQuantity DESC
        LIMIT :limit
    """)
    suspend fun getTopProducts(startDate: Long, endDate: Long, limit: Int = 10): List<TopProduct>

    // Ventas por método de pago
    @Query("""
        SELECT 
            pm.name as paymentMethodName,
            COUNT(*) as transactionCount,
            SUM(s.total) as totalAmount
        FROM sales s
        INNER JOIN payment_methods pm ON pm.id = s.paymentMethodId
        WHERE s.createdAt BETWEEN :startDate AND :endDate
        AND s.status = 'COMPLETED'
        GROUP BY s.paymentMethodId
    """)
    suspend fun getSalesByPaymentMethod(startDate: Long, endDate: Long): List<PaymentMethodSummary>

    // Ventas por hora (para análisis de horarios pico)
    @Query("""
        SELECT 
            strftime('%H', createdAt/1000, 'unixepoch', 'localtime') as hour,
            COUNT(*) as transactionCount,
            SUM(total) as totalRevenue
        FROM sales
        WHERE createdAt BETWEEN :startDate AND :endDate
        AND status = 'COMPLETED'
        GROUP BY hour
        ORDER BY hour
    """)
    suspend fun getSalesByHour(startDate: Long, endDate: Long): List<HourlySales>

    // Productos con stock bajo
    @Query("""
        SELECT * FROM products 
        WHERE stock <= minStock 
        AND isActive = 1
        ORDER BY stock ASC
    """)
    fun getLowStockProducts(): Flow<List<Product>>

    // Corte de caja
    @Query("""
        SELECT 
            pm.name as paymentMethodName,
            COUNT(*) as transactionCount,
            SUM(s.total) as totalAmount
        FROM sales s
        INNER JOIN payment_methods pm ON pm.id = s.paymentMethodId
        WHERE s.cashRegisterId = :cashRegisterId
        AND s.status = 'COMPLETED'
        GROUP BY s.paymentMethodId
    """)
    suspend fun getCashRegisterSummary(cashRegisterId: String): List<PaymentMethodSummary>
}

⚠️ 12. Reglas de Oro para POS
text

1. 💰 NUNCA usar Float/Double para dinero → usar BigDecimal o Long (centavos)

2. 🔌 OFFLINE FIRST → La app DEBE funcionar sin internet al 100%

3. 🔒 TRANSACCIONES ATÓMICAS → Venta + Descuento de inventario 
   + Registro en caja = TODO o NADA (usar @Transaction en Room)

4. 📋 AUDITORÍA → Registrar TODA operación con usuario, fecha y detalle
   (nunca borrar registros, usar soft delete)

5. 🖨️ RESPALDO DE IMPRESIÓN → Si falla la impresora, guardar el recibo
   como PDF y permitir reimpresión

6. 🔄 IDEMPOTENCIA → Las ventas deben tener ID único para evitar 
   duplicados en sincronización

7. 📱 BLOQUEO DE PANTALLA → Impedir que el dispositivo se apague 
   durante una transacción activa

8. 💾 BACKUP AUTOMÁTICO → Backup de DB local periódico

9. 🚫 NO PERDER VENTAS → Si hay error, guardar en cola de reintentos

10. 📊 NÚMEROS CORRELATIVOS → Los números de venta deben ser 
    secuenciales y sin huecos (requisito fiscal en muchos países)

11. 🏷️ SNAPSHOT DE DATOS → Al crear una venta, guardar copia del 
    precio/nombre del producto (si cambia después, la venta histórica 
    debe mantener los datos originales)

12. ⚡ RENDIMIENTO → La pantalla de venta debe responder en <100ms
    Usar índices en DB para búsquedas frecuentes
